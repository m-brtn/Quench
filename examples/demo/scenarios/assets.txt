# Assets (ctest: demo_assets_<backend>, a part of batch.txt): cooked models and textures load with a simulated latency
# of 0.2 s (real time), so the placeholders show first; the broken and the missing file fail and keep their magenta
# boxes; a reload swaps the same data in; moving the streaming focus loads the cells it comes near and releases the ones
# it leaves, and the 4 KB texture cache evicts what they released. The capture is taken at 1.5 s, with everything
# loaded.
section assets

set dither 0
# FXAA instead of TAA: the loads finish in real time, so the frame a texture arrives in varies between runs, and TAA's
# history keeps a trace of that (identical captures with FXAA, slightly different ones with TAA).
set aa fxaa
set latency 0.2
set move_focus 0
set focus_angle 0
expect assets_loading > 0
expect showcase_ready == 0
until assets_idle == 1 3
expect showcase_ready == 3
expect showcase_failed == 2
expect assets_failed == 2
expect streaming_active > 4

# A reload of everything in use: the reload counter moves, nothing fails.
set reload 1
until assets_reloads >= 6 2
until assets_idle == 1 2
expect assets_failed == 2
expect showcase_ready == 3

# The focus moves to the other side of the grid under a small texture cache.
set texture_budget 4096
set focus_angle 180
until streaming_unloads > 0 1
until assets_idle == 1 2
expect assets_evictions > 0
expect showcase_ready == 3
# Everything settled before the capture at 1.5 s (a slow machine fails here, not in the image comparison).
expect time < 1.3
until time >= 1.5 3
windows off
capture demo_assets.png
windows on
quit
