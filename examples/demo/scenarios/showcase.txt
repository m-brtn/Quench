# README gallery, captured offscreen at 1920x1080 on native Metal.
# Reproduction and image provenance: docs/images/CREDITS.md.
# This is a photo tour, separate from the regression checks and their golden images.
section weather
windows off
set dither 0
set quality full
set steps 128
set time_scale 0
set preset towering
set precipitation 0
set sun_elevation 3
set sun_azimuth 25
set weather_time 1200
camera 0 -25 600 0 650 -600
wait 2
expect clouds == 2
capture weather.png

section particles
windows off
set dither 0
set wind 3
set dusk 1
set density 1
set restart 1
wait 4.004
expect trail_vertices > 1500
expect particles > 5000
capture particles.png

section raytracing
windows off
set dither 0
until rt_ready == 1 5
set upscaling bilinear
set scale 1
set turn_sun 0
set walk 0
set spin 0
camera 24 14 7 0 .7 .1
wait 3
capture raytracing.png
quit
