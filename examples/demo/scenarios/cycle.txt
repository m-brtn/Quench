# Section switching (ctest: demo_cycle): three rounds through every section, a reset of each in the third.
# The first round creates every section (and waits for the assets of those that load some); after it, switching
# leaves the scene and the log as they are, the GPU resources within 3 (fences and transfer buffers of the frames
# and uploads in flight come and go), the memory grows by less than 32 MB (allocator noise; a section that leaked
# its scene at every switch would be far more), and a reset makes the same section again (the assets section logs
# its broken and missing model again, on purpose).

wait 0.1
section materials
wait 0.03
section shadows
wait 0.03
section sky
wait 0.03
section post
wait 0.03
section hdr
wait 0.03
section physics
wait 0.03
section input
wait 0.03
section assets
wait 0.03
until assets_idle == 1 3
section audio
wait 0.03
until sounds_ready == 1 3
section animation
wait 0.03
until anim_ready == 1 3
section raytracing
wait 0.03
until rt_ready == 1 3
section materials
wait 0.2
expect sections_ready == 11
mark gpu_resources
mark memory_mb
mark scene_instances
mark log_errors
section materials
wait 0.03
section shadows
wait 0.03
section sky
wait 0.03
section post
wait 0.03
section hdr
wait 0.03
section physics
wait 0.03
section input
wait 0.03
section assets
wait 0.03
section audio
wait 0.03
section animation
wait 0.03
section raytracing
wait 0.03
section materials
wait 0.2
expect d_gpu_resources < 4
expect d_gpu_resources > -4
expect d_scene_instances == 0
expect d_memory_mb < 32
expect d_log_errors == 0
section materials
reset_section
wait 0.03
section shadows
reset_section
wait 0.03
section sky
reset_section
wait 0.03
section post
reset_section
wait 0.03
section hdr
reset_section
wait 0.03
section physics
reset_section
wait 0.03
section input
reset_section
wait 0.03
section assets
reset_section
wait 0.03
until assets_idle == 1 3
section audio
reset_section
wait 0.03
until sounds_ready == 1 3
section animation
reset_section
wait 0.03
until anim_ready == 1 3
section raytracing
reset_section
wait 0.03
until rt_ready == 1 3
section materials
wait 0.3
expect sections_ready == 11
expect d_gpu_resources < 4
expect d_gpu_resources > -4
expect d_scene_instances == 0
expect d_memory_mb < 32
quit
