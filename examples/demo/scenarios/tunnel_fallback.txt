# The tunnel's fallbacks on SDL GPU (ctest: demo_tunnel_fallback_<backend>, a part of batch.txt): the eight area
# lights still light and shine (LTC: diffuse and specular) but cast no shadows, reflections are the night sky's,
# indirect light its ambient light, and the path tracer falls back to the real-time frame; the renderer names each
# fallback (Renderer::fallbacks). The capture is taken at 2.5 s from the overview, the lamps standing at their clock's
# start, 180 frames after a camera cut at 1 s, as in tunnel.txt.
section tunnel

set dither 0
until tunnel_ready == 1 10
reset_camera
set animate 0
set lamp_time 0
wait 0.1
expect drive == 0
expect lights == 10
expect area_lights == 8
expect local_shadows == 0
expect reflections == 0
expect indirect == 0
expect ray_instances == 0
expect upscaling == 0
expect render_scale == 1
expect fallbacks >= 3
set path_tracer 1
wait 0.05
expect path_tracer == 0
set path_tracer 0
set tubes 0
set portal 0
wait 0.05
expect area_lights == 5
set tubes 1
set portal 1
wait 0.05
expect area_lights == 8
# A second cut at a fixed time: the frames from it to the capture do not depend on how long the textures took.
until time >= 1.0042 3
reset_camera
until time >= 2.5042 3
windows off
capture demo_tunnel.png
windows on
quit
