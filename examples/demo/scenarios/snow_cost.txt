# The cost of the snow per quality setting (docs/performance.md, "Snow"): the weather section's Snowfall preset from
# the ground (the flakes, the cover on the surfaces, the occlusion map and the shafts of the snow beyond the flakes) and
# its Snow showers preset from the sky view (the shafts of the showers, one of them over the camera), with the rain's
# drawing off, half-size clouds, the weather's clock stopped, 3 s per setting after 1 s to settle; the GPU busy time
# per range comes from the line after each perf line (rain_map: the occlusion map and the weather maps' regions; rain:
# the copy of the image the flakes are drawn over; snow: the flakes; rain_shafts: the shafts; hdr: the main pass, where
# the surfaces whiten). Offscreen, at any size, for example on the native Metal backend:
#   quench-demo --gpu-backend metal --window 1920x1080 --audio-device null --script snow_cost.txt
section weather
set dither 0
set quality reduced
set rain_shafts 1
set rainbow 0
set rain off
set view ground
set preset snowfall
set weather_time 4000
set time_scale 0

set snow off
wait 1
perf_begin snow_off
wait 3
perf_end

set snow low
wait 1
perf_begin snow_low
wait 3
perf_end

set snow medium
wait 1
perf_begin snow_medium
wait 3
perf_end

set snow high
wait 1
perf_begin snow_high
wait 3
perf_end

# The phone defaults: the rain's and the snow's low quality, without the shafts.
set rain low
set snow low
set rain_shafts 0
wait 1
perf_begin snow_phone
wait 3
perf_end
set rain off
set rain_shafts 1

# Distant snow showers from the sky view: the shafts' march.
set view sky
set preset snow_showers
set weather_time 4000
set time_scale 0
set snow off
wait 1
perf_begin snow_sky_off
wait 3
perf_end

set snow low
wait 1
perf_begin snow_sky_low
wait 3
perf_end

set snow medium
wait 1
perf_begin snow_sky_medium
wait 3
perf_end

set snow high
wait 1
perf_begin snow_sky_high
wait 3
perf_end
quit
