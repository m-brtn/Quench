# Screen-space ambient occlusion on a phone (AmbientOcclusion::ScreenSpace; docs/performance.md, "Screen-space
# ambient occlusion") on SDL GPU: the shadows section, the ray tracing section and the tunnel
# without occlusion and with GTAO, at the display's rate in 4 s regions, each section from the nominal thermal state;
# then a capture of the ray tracing section with GTAO:
#   copy this file into the app's user directory, launch with --gpu-backend sdl --script gtao_phone.txt
# On the native backend GTAO runs too (set ao screen asks for it explicitly).

section shadows
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set ao none
wait 0.5
perf_begin shadows_no_ao
wait 4
perf_end
set ao screen
wait 0.5
expect screen_occlusion == 1
perf_begin shadows_gtao
wait 4
perf_end
set ao none

section raytracing
until rt_ready == 1 5
set limit_60hz 0
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set ao none
wait 0.5
perf_begin rt_no_ao
wait 4
perf_end
set ao screen
wait 0.5
expect screen_occlusion == 1
perf_begin rt_gtao
wait 4
perf_end
windows off
capture gtao_phone.png
windows on
set ao none

section tunnel
until tunnel_ready == 1 10
set limit_60hz 0
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set ao none
wait 0.5
perf_begin tunnel_no_ao
wait 4
perf_end
set ao screen
wait 0.5
expect screen_occlusion == 1
perf_begin tunnel_gtao
wait 4
perf_end
set ao none
quit
