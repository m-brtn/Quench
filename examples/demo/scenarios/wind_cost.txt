# The cost of the wind (docs/performance.md, "Wind"): the weather section's meadow (79 swaying plants) in the windy
# storm, the weather's clock stopped, 3 s per setting after 1 s to settle; the GPU busy time per range comes from the
# line after each perf line (hdr: the main pass, where the plants sway in their vertex shader; shadows: the cascades,
# which sway them too; rain: the streaks and splashes, carried by the gusts; snow: the flakes). The plants without
# wind (the sway returns at once) against the gale; the rain and the snow in the gale without gusts and with them.
# Lightning costs nothing on the GPU here (a branch in the sky; on the CPU the frame's flashes). Offscreen, at any size,
# for example on the native Metal backend:
#   quench-demo --gpu-backend metal --window 1920x1080 --audio-device null --script wind_cost.txt
section weather
set dither 0
set quality reduced
set rain_shafts 1
set rainbow 0
set rain off
set snow off
set view meadow
set preset windy_storm
set lightning 0
set weather_time 4000
set time_scale 0

# The sway alone, the rain's drawing off.
set wind_speed 0
wait 1
perf_begin sway_off
wait 3
perf_end

set wind_speed 22
wait 1
perf_begin sway_on
wait 3
perf_end

# The rain in the gale, without gusts and with the preset's.
set rain medium
set gusts 0
set turbulence 0
wait 1
perf_begin rain_steady
wait 3
perf_end

set gusts 0.6
set turbulence 0.4
wait 1
perf_begin rain_gusts
wait 3
perf_end

# The snow in the gale.
set rain off
set snow medium
set temperature -6
set gusts 0
set turbulence 0
wait 1
perf_begin snow_steady
wait 3
perf_end

set gusts 0.6
set turbulence 0.4
wait 1
perf_begin snow_gusts
wait 3
perf_end

# The phone defaults: the rain's low quality without the shafts, with the gusts.
set snow low
set rain low
set rain_shafts 0
set temperature 15
wait 1
perf_begin wind_phone
wait 3
perf_end
quit
