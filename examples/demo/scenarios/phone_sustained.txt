# The phone defaults of the ray tracing section and of the tunnel held for ten minutes at 60 Hz (docs/performance.md,
# "Ten minutes on the iPhone"): five minutes of each in 20 s regions, from the nominal thermal state and without a
# pause between them; each region's perf line carries its thermal state:
#   copy this file into the app's user directory, launch with --gpu-backend metal --script phone_sustained.txt

section raytracing
until rt_ready == 1 5
expect limit_60hz == 1
expect max_frame_rate == 60

# Cool down: the cheapest frame, few of them (as in raytracing_phone.txt); reset_section brings the defaults back.

set ao none
set local_shadows 0
set reflections sky
set upscaling bilinear
max_frame_rate 20
until thermal_state <= 0 300
reset_section
until rt_ready == 1 5
