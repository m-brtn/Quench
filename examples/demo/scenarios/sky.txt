# Sky and fog (ctest: demo_sky_metal, demo_sky_vulkan): the daylight palette, the atmosphere model and its gradient
# fallback, no sky, fog. The capture is taken at 2 s with the start state (sun 9 degrees high, fog 0.006 /m) turned to
# azimuth 15 degrees, so the sun is in view: at this height the disc and the inner glow are both above SDR white (the
# disc is never darker than the glow; render.renderer_sun_disc_outshines_its_glow checks it in SDR and HDR).

set dither 0
wait 0.2
expect atmosphere == 1
expect fog_density > 0.005
set sun_elevation 60
wait 0.05
expect exposure < 1.1
expect shadow_draws > 0
# Night: no sun light, no shadows, more exposure.
set sun_elevation -10
wait 0.05
expect exposure > 2
expect shadow_draws == 0
set sun_elevation 9
set atmosphere 0
wait 0.05
expect atmosphere == 0
set atmosphere 1
set sky 0
set fog 0
wait 0.05
expect fog_density == 0

set sky 1
set fog 0.006
set sun_azimuth 15
wait 0.1
expect fog_density > 0.005
until time >= 2.1 3
quit
