# Frame-time benchmark of the engine demo (docs/performance.md): each rendering section, physics, assets, audio,
# animation, ray tracing and the tunnel for 5 s, at a fixed frame size and the default vsync:
#   quench-demo --frame-size 1920x1080 --script examples/demo/scenarios/benchmark.txt
# Every perf_end logs a perf line with the frame, CPU and GPU times and the section's draw counts. The perf regions run
# in real time like normal play (5 s: about 300 frames at 60 Hz, 600 at 120 Hz).

section materials
wait 1
perf_begin materials
wait 5
perf_end
expect perf_frames >= 150

section shadows
wait 1
perf_begin shadows
wait 5
perf_end
expect perf_frames >= 150

section sky
wait 1
perf_begin sky
wait 5
perf_end
expect perf_frames >= 150

section post
wait 1
perf_begin post
wait 5
perf_end
expect perf_frames >= 150

# The HDR section in SDR, then with HDR output (an Rgba16Float frame and the UI composite pass; on an SDR display the
# headroom is 1, so the cost is measured, not the look).
section hdr
wait 1
perf_begin hdr
wait 5
perf_end
expect perf_frames >= 150
hdr on
until hdr == 1 2
wait 1
perf_begin hdr_on
wait 5
perf_end
expect perf_frames >= 150
hdr off
until hdr == 0 2

section physics
wait 0.5
tap P
wait 0.5
perf_begin physics
wait 5
perf_end
expect perf_frames >= 150

# Streaming around the moving focus with the default budgets, after the first loads have finished.
section assets
until assets_idle == 1 3
wait 1
perf_begin assets
wait 5
perf_end
expect perf_frames >= 150

# Three 3D emitters, the tone generator and streamed music on the audio device (automated runs: --audio-device null).
section audio
until sounds_ready == 1 3
set tone 1
wait 1
perf_begin audio
wait 5
perf_end
expect perf_frames >= 150

# A crowd of 256 walking characters, the hero and the foot placement stand (259 characters, 20 joints each): their
# update on the job system and the compute skinning pass.
section animation
until anim_ready == 1 3
set crowd 256
until anim_ready == 1 3
wait 1
perf_begin animation
wait 5
perf_end
expect perf_frames >= 150

# Ray-traced shadows and occlusion where gpu::Caps allow them (the native Metal backend), else their fallbacks;
# raytracing_benchmark.txt measures the options one by one.
section raytracing
until rt_ready == 1 5
wait 1
perf_begin raytracing
wait 5
perf_end
expect perf_frames >= 150

# Eight area lights and two spots: traced lamp shadows and reflections where gpu::Caps allow them, else the lamps
# unshadowed (raster LTC shading) and the sky's reflections. From the overview with the lamps standing still (the
# numbers stay comparable with those from before they moved; raytracing_benchmark.txt measures them moving).
section tunnel
until tunnel_ready == 1 5
reset_camera
set animate 0
set lamp_time 0
wait 1
perf_begin tunnel
wait 5
perf_end
expect perf_frames >= 150
quit
