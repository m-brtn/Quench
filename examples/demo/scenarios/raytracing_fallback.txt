# Ray tracing's fallbacks on SDL GPU (ctest: demo_raytracing_metal, demo_raytracing_vulkan): the section asks for
# ray-traced shadows and ambient occlusion and for MetalFX, and gets the cascaded shadow maps, no occlusion and
# bilinear scaling. The capture is taken at 2.5 s, as in raytracing.txt.

set dither 0
until rt_ready == 1 5
wait 0.1
expect sun_shadows == 0
expect ambient_occlusion == 0
expect cascades == 3
expect shadow_draws > 0
expect ray_instances == 0
set upscaling temporal
set scale 0.5
wait 0.05
expect upscaling == 0
expect render_scale == 0.5
set upscaling spatial
wait 0.05
expect upscaling == 0
set upscaling bilinear
set scale 1
wait 0.05
expect render_scale == 1
until time >= 2.6 3
quit
