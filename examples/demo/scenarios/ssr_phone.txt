# Screen-space reflections on a phone (Reflections::ScreenSpace; docs/performance.md, "Screen-space reflections") on
# SDL GPU: the materials section, the ray tracing section and the tunnel with the sky's
# reflections and with screen-space ones, at the display's rate in 4 s regions, each section from the nominal thermal
# state; then a capture of the ray tracing section with them:
#   copy this file into the app's user directory, launch with --gpu-backend sdl --script ssr_phone.txt
# On the native backend they run too (set reflections screen asks for them in place of the traced ones).

section materials
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set reflections sky
wait 0.5
perf_begin materials_sky
wait 4
perf_end
set reflections screen
wait 0.5
expect screen_reflections == 1
perf_begin materials_ssr
wait 4
perf_end
set reflections sky

section raytracing
until rt_ready == 1 5
set limit_60hz 0
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set reflections sky
wait 0.5
perf_begin rt_sky
wait 4
perf_end
set reflections screen
wait 0.5
expect screen_reflections == 1
perf_begin rt_ssr
wait 4
perf_end
windows off
capture ssr_phone.png
windows on
set reflections rt

section tunnel
until tunnel_ready == 1 10
set limit_60hz 0
max_frame_rate 20
until thermal_state <= 0 300
max_frame_rate 0
set reflections sky
wait 0.5
perf_begin tunnel_sky
wait 4
perf_end
set reflections screen
wait 0.5
expect screen_reflections == 1
perf_begin tunnel_ssr
wait 4
perf_end
set reflections rt
quit
